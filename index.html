function spawnRamp(zPos) { const lane = LANES[Math.floor(Math.random() * LANES.length)]; const shape = new THREE.Shape(); shape.moveTo(0, 0); shape.lineTo(0, 1.5); shape.lineTo(6, 0); shape.closePath(); const extrudeSettings = { depth: 2.0, bevelEnabled: false }; const rampGeo = new THREE.ExtrudeGeometry(shape, extrudeSettings); const rampMat = new THREE.MeshLambertMaterial({ color: 0xffa502 }); const ramp = new THREE.Mesh(rampGeo, rampMat); ramp.rotation.y = Math.PI / 2; ramp.position.set(lane - 1.0, 0, zPos); ramp.userData = { type: 'ramp' }; scene.add(ramp); obstacles.push(ramp); } function spawnTunnel(zPos) { // Create an architectural frame archway obstacle structure const tunnelGroup = new THREE.Group(); const mat = new THREE.MeshLambertMaterial({ color: 0x718093 }); // Left pillar const leftCol = new THREE.Mesh(new THREE.BoxGeometry(1.5, 6, 3), mat); leftCol.position.set(-5.5, 3, 0); // Right pillar const rightCol = new THREE.Mesh(new THREE.BoxGeometry(1.5, 6, 3), mat); rightCol.position.set(5.5, 3, 0); // Top roof beam spanning across const topBeam = new THREE.Mesh(new THREE.BoxGeometry(13, 1.5, 3), mat); topBeam.position.set(0, 6.75, 0); tunnelGroup.add(leftCol); tunnelGroup.add(rightCol); tunnelGroup.add(topBeam); tunnelGroup.position.set(0, 0, zPos); tunnelGroup.userData = { type: 'tunnel' }; scene.add(tunnelGroup); obstacles.push(tunnelGroup); } function spawnCoin(zPos, lane, height = 0.8) { const coinGeo = new THREE.CylinderGeometry(0.35, 0.35, 0.1, 16); const coinMat = new THREE.MeshLambertMaterial({ color: 0xffd700 }); const coin = new THREE.Mesh(coinGeo, coinMat); coin.rotation.x = Math.PI / 2; coin.position.set(lane, height, zPos); scene.add(coin); coins.push(coin); } function createExplosion(position) { const particleCount = 20; const geometry = new THREE.BoxGeometry(0.3, 0.3, 0.3); for (let i = 0; i < particleCount; i++) { const material = new THREE.MeshBasicMaterial({ color: Math.random() > 0.5 ? 0xff4757 : 0xffa502 }); const particle = new THREE.Mesh(geometry, material); particle.position.copy(position); particle.userData = { vx: (Math.random() - 0.5) * 12, vy: Math.random() * 10 + 2, vz: (Math.random() - 0.5) * 12, life: 1.0 }; scene.add(particle); particles.push(particle); } } // SPAWN OBSTACLE HELPER function spawnRandomObstacle(z) { const rand = Math.random(); if (rand < 0.45) spawnTrain(z); else if (rand < 0.75) spawnRamp(z); else spawnTunnel(z); } // INITIAL SPAWNS for (let z = -30; z > -250; z -= 40) { spawnRandomObstacle(z); const coinLane = LANES[Math.floor(Math.random() * LANES.length)]; for (let c = 0; c < 2; c++) { spawnCoin(z - (c * 3), coinLane); } } // INPUT LOGIC document.addEventListener('keydown', (e) => { if (isGameOver) return; if ((e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') && currentLane > 0) { currentLane--; } if ((e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') && currentLane < LANES.length - 1) { currentLane++; } if ((e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') && isGrounded) { velocityY = jumpPower; isGrounded = false; } if (e.key === ' ' && !isShieldActive && !shieldCooldown) { activateShield(); } }); function activateShield() { isShieldActive = true; shieldMesh.visible = true; document.getElementById('shield-status').innerText = "Shield [SPACEBAR]: ACTIVE!"; document.getElementById('shield-status').style.color = "#ffdd59"; } function deactivateShield() { isShieldActive = false; shieldMesh.visible = false; shieldCooldown = true; isInvulnerable = true; setTimeout(() => { isInvulnerable = false; }, 300); document.getElementById('shield-status').innerText = "Shield [SPACEBAR]: RECHARGING..."; document.getElementById('shield-status').style.color = "#ff4757"; setTimeout(() => { shieldCooldown = false; document.getElementById('shield-status').innerText = "Shield [SPACEBAR]: READY"; document.getElementById('shield-status').style.color = "#00d2d3"; }, 5000); } // MAIN LOOP function animate() { if (isGameOver) return; requestAnimationFrame(animate); const delta = clock.getDelta(); speed += 0.5 * delta; score += Math.floor(speed * delta * 5); document.getElementById('score').innerText = score; player.position.x += (LANES[currentLane] - player.position.x) * (15 * delta); player.position.y += velocityY * delta; velocityY += gravity * delta; if (player.position.y <= BASE_Y) { player.position.y = BASE_Y; velocityY = 0; isGrounded = true; } if (isShieldActive) { shieldMesh.rotation.y += 2 * delta; } // Update Particles for (let i = particles.length - 1; i >= 0; i--) { const p = particles[i]; p.position.x += p.userData.vx * delta; p.position.y += p.userData.vy * delta; p.position.z += p.userData.vz * delta; p.userData.life -= 1.5 * delta; p.scale.setScalar(Math.max(0, p.userData.life)); if (p.userData.life <= 0) { scene.remove(p); particles.splice(i, 1); } } // Collisions const playerBox = new THREE.Box3().setFromObject(player); playerBox.expandByScalar(-0.1); for (let i = obstacles.length - 1; i >= 0; i--) { const obs = obstacles[i]; obs.position.z += speed * delta; const obsBox = new THREE.Box3().setFromObject(obs); if (playerBox.intersectsBox(obsBox)) { if (obs.userData.type === 'ramp') { velocityY = jumpPower * 1.1; isGrounded = false; } else if (obs.userData.type === 'train' || obs.userData.type === 'tunnel') { if (isShieldActive) { createExplosion(obs.position); scene.remove(obs); obstacles.splice(i, 1); deactivateShield(); continue; } else if (!isInvulnerable) { triggerGameOver(); return; } } } if (obs.position.z > 10) { scene.remove(obs); obstacles.splice(i, 1); spawnRandomObstacle(-230); } } // Coin Collisions for (let i = coins.length - 1; i >= 0; i--) { const coin = coins[i]; coin.position.z += speed * delta; coin.rotation.z += 3 * delta; const coinBox = new THREE.Box3().setFromObject(coin); if (playerBox.intersectsBox(coinBox)) { scene.remove(coin); coins.splice(i, 1); coinsCollected++; document.getElementById('coins').innerText = coinsCollected; } else if (coin.position.z > 10) { scene.remove(coin); coins.splice(i, 1); if (Math.random() < 0.5) { const lane = LANES[Math.floor(Math.random() * LANES.length)]; spawnCoin(-230, lane); } } } renderer.render(scene, camera); } function triggerGameOver() { isGameOver = true; document.getElementById('final-score').innerText = score; document.getElementById('final-coins').innerText = coinsCollected; document.getElementById('game-over').style.display = 'block'; } function resetGame() { obstacles.forEach(obs => scene.remove(obs)); coins.forEach(coin => scene.remove(coin)); particles.forEach(p => scene.remove(p)); obstacles = []; coins = []; particles = []; score = 0; coinsCollected = 0; speed = INITIAL_SPEED; currentLane = 1; player.position.set(LANES[1], BASE_Y, 0); velocityY = 0; isGrounded = true; isGameOver = false; isInvulnerable = false; isShieldActive = false; shieldCooldown = false; shieldMesh.visible = false; document.getElementById('shield-status').innerText = "Shield [SPACEBAR]: READY"; document.getElementById('shield-status').style.color = "#00d2d3"; document.getElementById('score').innerText = '0'; document.getElementById('coins').innerText = '0'; document.getElementById('game-over').style.display = 'none'; clock.start(); for (let z = -30; z > -250; z -= 40) { spawnRandomObstacle(z); const coinLane = LANES[Math.floor(Math.random() * LANES.length)]; spawnCoin(z, coinLane); } animate(); } window.addEventListener('resize', () => { camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix(); renderer.setSize(window.innerWidth, window.innerHeight); }); animate();
